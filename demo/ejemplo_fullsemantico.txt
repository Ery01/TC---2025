// ERROR: Declarar 2 veces una función 'duplicateFunc' (redifinición).

char duplicateFunc() { return 'a'; } // <-- Este debería ser un ERROR

// Función con parámetros para probar errores
int add(int num1, int num2) {
    return num1 + num2;
}

// ERROR: Pasar parámetros no esperados (demasiados).
int sumResult1 = add(1, 2, 3); // <-- ERROR: Demasiados argumentos para 'add'

// ERROR: Pasar parámetros no esperados (faltan).

// ERROR: Pasar parámetros no esperados (tipo incorrecto).
int sumResult3 = add(1, 'b'); // <-- ERROR: Tipo de argumento incompatible para 'add' (char en lugar de int)
